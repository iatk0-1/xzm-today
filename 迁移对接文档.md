# xzm-today 云开发迁移到后端接口对接文档

## 一、项目概述

### 1.1 迁移目标
- **原架构**: 微信小程序云开发 (wx.cloud) + 云数据库
- **新架构**: 微信小程序 + Spring Boot 后端 API (localhost:8080)
- **核心变更**: 移除所有云开发代码，改用 REST API 调用后端接口

### 1.2 后端接口基础信息
| 配置项 | 值 |
|--------|-----|
| 基础 URL | `http://localhost:8080` |
| API 前缀 | `/api/v1` |
| 认证方式 | JWT Token (Bearer) |
| 小程序 AppID | `wx7692230ac8ecdaa8` |

### 3.2 api.js - API 请求封装
```javascript
// miniprogram/utils/api.js
const config = require('./config');

/**
 * 获取存储的 Token
 */
function getToken() {
  return wx.getStorageSync(config.TOKEN_KEY) || '';
}

/**
 * 通用请求封装
 */
function request(options) {
  return new Promise((resolve, reject) => {
    const token = getToken();

    wx.request({
      url: config.API_BASE_URL + options.url,
      method: options.method || 'GET',
      data: options.data || {},
      header: {
        'Content-Type': 'application/json',
        'Authorization': token ? `Bearer ${token}` : ''
      },
      success: (res) => {
        if (res.statusCode === 200 || res.statusCode === 201) {
          resolve(res.data);
        } else if (res.statusCode === 401) {
          // Token 过期，需要刷新或重新登录
          wx.showToast({ title: '登录已过期', icon: 'none' });
          wx.clearStorageSync();
          wx.reLaunch({ url: '/pages/user/user' });
          reject({ error: 'UNAUTHORIZED' });
        } else {
          reject(res.data);
        }
      },
      fail: (err) => {
        wx.showToast({ title: '网络请求失败', icon: 'none' });
        reject(err);
      }
    });
  });
}

module.exports = {
  get: (url, data) => request({ url, method: 'GET', data }),
  post: (url, data) => request({ url, method: 'POST', data }),
  put: (url, data) => request({ url, method: 'PUT', data }),
  patch: (url, data) => request({ url, method: 'PATCH', data }),
  delete: (url, data) => request({ url, method: 'DELETE', data })
};
```

### 3.3 auth.js - 认证管理
```javascript
// miniprogram/utils/auth.js
const api = require('./api');
const config = require('./config');

/**
 * 小程序登录
 */
async function login(nickname = '') {
  // 先获取微信 code
  const loginRes = await wxLogin();

  // 调用后端登录接口
  const loginData = {
    code: loginRes.code,
    nickname: nickname
  };

  const res = await api.post('/auth/miniapp/login', loginData);

  // 保存 token 和用户信息
  wx.setStorageSync(config.TOKEN_KEY, res.accessToken);
  wx.setStorageSync(config.REFRESH_TOKEN_KEY, res.refreshToken);
  wx.setStorageSync(config.USER_INFO_KEY, {
    userId: res.userId,
    openid: res.openid,
    phone: res.phone,
    nickname: res.nickname
  });

  return res;
}

/**
 * 获取微信登录 code
 */
function wxLogin() {
  return new Promise((resolve, reject) => {
    wx.login({
      timeout: 10000,
      success: resolve,
      fail: reject
    });
  });
}

/**
 * 检查登录状态
 */
function checkLogin() {
  const token = wx.getStorageSync(config.TOKEN_KEY);
  const userInfo = wx.getStorageSync(config.USER_INFO_KEY);
  return !!(token && userInfo);
}

/**
 * 获取用户信息
 */
function getUserInfo() {
  return wx.getStorageSync(config.USER_INFO_KEY) || null;
}

/**
 * 退出登录
 */
function logout() {
  wx.clearStorageSync();
}

module.exports = {
  login,
  checkLogin,
  getUserInfo,
  logout
};
```

// miniprogram/pages/detail/detail.js
const api = require('../../utils/api');

Page({
  data: {
    // ... 保持不变
  },

  onLoad: function(options) {
    // ... 系统信息获取保持不变
    const productId = options.id;

    if (productId) {
      this.getProductDetail(productId);
    }
  },

  // 改造：获取商品详情
  getProductDetail: async function(id) {
    wx.showLoading({ title: '加载中...' });

    try {
      const res = await api.get(`/products/${id}`);
      // 后端返回格式：{ product: {...}, skus: [...] }
      const product = res.product;
      const skus = res.skus || [];

      // 转换 skuMatrix 格式以兼容现有代码
      const skuMatrix = skus.map(sku => {
        const specParts = sku.spec ? sku.spec.split('/') : ['默认', '均码'];
        return {
          color: specParts[0] || '默认',
          size: specParts[1] || '均码',
          price: sku.retailPrice,
          stock: sku.stockMain
        };
      });

      this.setData({
        product: {
          ...product,
          skuMatrix: skuMatrix,
          // 字段映射
          image: product.coverUrl,
          bannerImgs: product.bannerImages || [],
          lookbookImgs: product.lookbookImages || [],
          detailImgs: product.detailImages || [],
          manualRelatedIds: product.relatedProductIds || []
        },
        // 处理轮播图
        bannerImgs: product.coverUrl
          ? [product.coverUrl, ...(product.bannerImages || [])]
          : (product.bannerImages || [])
      });

      // 获取关联商品
      if (product.relatedProductIds && product.relatedProductIds.length > 0) {
        this.getRelatedProducts(product.relatedProductIds);
      }
    } catch (err) {
      wx.hideLoading();
      wx.showToast({ title: '商品不存在', icon: 'none' });
    }
  },

  // 改造：获取关联商品
  getRelatedProducts: async function(ids) {
    try {
      const res = await api.get('/products/related', {
        ids: ids.join(','),
        limit: 4
      });
      this.setData({ relatedProducts: res });
    } catch (err) {
      console.error('获取关联商品失败:', err);
    }
  },

  // ... 其他方法保持不变

#### 4.4.2 修改后的 checkout.js
```javascript
// miniprogram/pages/checkout/checkout.js
const api = require('../../utils/api');
const auth = require('../../utils/auth');

Page({
  data: {
    address: null,
    checkoutItems: [],
    totalPrice: 0
  },

  onLoad: function() {
    let items = wx.getStorageSync('checkoutItems') || [];
    let total = 0;
    items.forEach(item => {
      let currentPrice = Number(item.finalPrice || item.price || 0);
      total += (currentPrice * item.count);
    });

    this.setData({
      checkoutItems: items,
      totalPrice: total.toFixed(2)
    });
  },

  chooseAddress: function() {
    // 保持微信原生地址选择
    wx.chooseAddress({
      success: (res) => {
        this.setData({
          address: {
            recipient: res.userName,
            phone: res.telNumber,
            province: res.provinceName,
            city: res.cityName,
            district: res.countyName,
            detail: res.detailInfo
          }
        });
      }
    });
  },

  submitOrder: async function() {
    const { address, checkoutItems, totalPrice } = this.data;

    if (!address) {
      wx.showToast({ title: '请先选择收货地址', icon: 'none' });
      return;
    }

    wx.showLoading({ title: '创建订单...' });

    try {
      // 1. 创建订单 - 构造后端要求的格式
      const orderItems = checkoutItems.map(item => ({
        skuId: item.skuId || 0,  // 需要从本地 SKU 映射
        qty: item.count,
        salePrice: Number(item.finalPrice || item.price),
        pool: 'main'  // 默认使用主库存
      }));

      const createOrderData = {
        items: orderItems
        // userId 可选，后端从 Token 中获取
      };

      const orderRes = await api.post('/orders', createOrderData);
      const orderId = orderRes.id;

      // 2. 调用微信支付预下单
      wx.showLoading({ title: '准备支付...' });
      const payRes = await api.post(`/orders/${orderId}/pay/wechat`);

      // 3. 拉起微信支付
      if (payRes && payRes.packageValue) {
        wx.requestPayment({
          timeStamp: payRes.timeStamp,
          nonceStr: payRes.nonceStr,
          package: payRes.packageValue,
          signType: payRes.signType,
          paySign: payRes.paySign,
          success: (successRes) => {
            wx.showToast({ title: '支付成功!', icon: 'success' });
            wx.removeStorageSync('cart');
            wx.removeStorageSync('checkoutItems');
            setTimeout(() => {
              wx.switchTab({ url: '/pages/index/index' });
            }, 1500);
          },
          fail: (err) => {
            if (err.errMsg === 'requestPayment:fail cancel') {
              wx.showToast({ title: '您手动取消了支付', icon: 'none' });
            } else {
              wx.showModal({
                title: '支付失败',
                content: err.errMsg,
                showCancel: false
              });
            }
          }
        });
      } else {
        wx.showModal({
          title: '支付准备失败',
          content: '未能获取支付参数',
          showCancel: false
        });
      }
    } catch (err) {
      wx.hideLoading();
      wx.showModal({
        title: '订单创建失败',
        content: JSON.stringify(err),
        showCancel: false
      });
    }
  }
});
```

const api = require('../../utils/api');

// 状态映射
const STATUS_MAP = {
  '全部': null,
  '待付款': 'PENDING_PAYMENT',
  '待发货': 'PAID',
  '已发货': 'SHIPPED',
  '已完成': 'COMPLETED',

    currentTab: '全部',

      'pay': '待付款',
      'ship': '待发货',
      'receive': '已发货',
      'done': '已完成',
      'refund': '退款/售后',
      'closed': '已关闭'

      this.loadOrders();
    });

  loadOrders: async function() {

    wx.showLoading({ title: '加载订单中...' });

    try {
      const backendStatus = STATUS_MAP[this.data.currentTab];
      const params = {};
      if (backendStatus) {
        params.status = backendStatus;
      }
      params.page = 1;
      params.size = 20;

      const res = await api.get('/orders', params);

      // 后端返回格式：{ items: [...], total: N, page: 1, size: 20 }
      const orders = res.items || [];
      this.setData({ orders, isLoading: false });
    } catch (err) {
      wx.hideLoading();
      wx.showToast({ title: '获取订单失败', icon: 'none' });
      console.error(err);
    }
  },

  // 改造：确认收货

          } catch (err) {

            wx.showToast({ title: '操作失败', icon: 'none' });
          }
        }

Page({

    wishes: []
  },

  onShow: function() {
    this.loadWishes();
  },

  // 改造：获取心愿列表
  loadWishes: async function() {
    wx.showLoading({ title: '探索中...' });

    try {
      const res = await api.get('/wishes');
      // 后端返回格式：{ wishes: [...] }
      this.setData({ wishes: res.wishes || res, isLoading: false });
    } catch (err) {
      wx.hideLoading();
      console.error('获取心愿失败:', err);

  // 改造：点赞
  handleLike: async function(e) {
    const index = e.currentTarget.dataset.index;
    const wishId = e.currentTarget.dataset.id;
    let currentWishes = this.data.wishes;
    let targetWish = currentWishes[index];

    targetWish.isLiked = !targetWish.isLiked;
    if (targetWish.isLiked) {
      targetWish.likes = (targetWish.likes || 0) + 1;
    } else {
      targetWish.likes = (targetWish.likes || 0) - 1;
      if (targetWish.likes < 0) targetWish.likes = 0;
    }

    this.setData({ wishes: currentWishes });

    try {
      if (targetWish.isLiked) {
        await api.post(`/wishes/${wishId}/like`);
      } else {
        await api.post(`/wishes/${wishId}/unlike`);
      }
    } catch (err) {
      // 回滚状态
      targetWish.isLiked = !targetWish.isLiked;
      if (targetWish.isLiked) {
        targetWish.likes = (targetWish.likes || 0) + 1;
      } else {
        targetWish.likes = (targetWish.likes || 0) - 1;
      }

// miniprogram/pages/admin/admin.js
const api = require('../../utils/api');

Page({
  // ... data 保持不变

  // 改造：视频上传
  uploadVideo() {
    wx.chooseMedia({
      count: 1,
      mediaType: ['video'],
      success: (res) => {
        const tempFilePath = res.tempFiles[0].tempFilePath;
        wx.showLoading({ title: '上传视频中...' });

        this.uploadFile(tempFilePath, 'video/mp4').then(url => {
          wx.hideLoading();
          wx.showToast({ title: '视频上传成功', icon: 'success' });
          this.setData({ videoUrl: url });
        }).catch(err => {
          wx.hideLoading();
          wx.showModal({ title: '上传失败', content: err.errMsg, showCancel: false });
        });
      }
    });
  },

  // 改造：通用文件上传
  uploadFile: function(filePath, contentType) {
    return new Promise((resolve, reject) => {
      wx.uploadFile({
        url: 'http://localhost:8080/api/v1/admin/files/upload',
        filePath: filePath,
        name: 'file',
        formData: {
          dir: 'products'
        },
        success: (res) => {
          const data = JSON.parse(res.data);
          resolve(data.url);
        },
        fail: (err) => {
          reject(err);
        }
      });
    });
  },

  // 改造：提交商品
  submitProduct: async function() {
    const { mediaList, title, selectedTags, skuList, lookbookImgs, detailImgs, manualRelated } = this.data;

    if (mediaList.length === 0 || !title || skuList.length === 0) {
      return wx.showToast({ title: '首图/名称/尺码颜色不能为空', icon: 'none' });
    }

    // 计算价格范围
    let prices = skuList.map(item => Number(item.price)).filter(p => !isNaN(p) && p > 0);
    if (prices.length === 0) {
      return wx.showToast({ title: '请填写正确的 SKU 价格', icon: 'none' });
    }
    let minPrice = Math.min(...prices);
    let maxPrice = Math.max(...prices);
    let displayPrice = (minPrice === maxPrice) ? String(minPrice) : `${minPrice} - ${maxPrice}`;

    wx.showLoading({ title: '上传中...', mask: true });

    try {
      // 1. 上传所有图片
      const uploadedMediaUrls = await this.uploadMediaList(mediaList);
      const uploadedLookbookUrls = await this.uploadImageList(lookbookImgs);
      const uploadedDetailUrls = await this.uploadImageList(detailImgs);

      // 2. 构造后端要求的 SKU 格式
      const skus = skuList.map(sku => ({
        spec: `${sku.color}/${sku.size}`,
        barcode: '',  // 可选
        retailPrice: Number(sku.price),
        stockMain: Number(sku.stock) || 0
      }));

      // 3. 构造创建商品请求
      const productData = {
        name: title,
        coverUrl: uploadedMediaUrls[0],  // 第一张是封面
        bannerImages: uploadedMediaUrls.slice(1),  // 其余的是轮播图
        tags: selectedTags,
        status: 'on',
        retailPrice: minPrice,  // 使用最低价作为零售价
        displayPrice: displayPrice,
        lookbookImages: uploadedLookbookUrls,
        detailImages: uploadedDetailUrls,
        relatedProductIds: manualRelated.map(item => item.id),
        skus: skus
      };

      wx.showLoading({ title: '创建商品...', mask: true });

      // 4. 调用后端创建商品 API
      await api.post('/products', productData);

      wx.hideLoading();
      wx.showToast({ title: '上架成功!', icon: 'success' });

      setTimeout(() => {
        wx.navigateBack();
      }, 1500);

    } catch (err) {
      wx.hideLoading();
      wx.showModal({
        title: '创建失败',
        content: JSON.stringify(err),
        showCancel: false
      });
    }
  },

  // 上传媒体列表
  uploadMediaList: async function(list) {
    const urls = [];
    for (const item of list) {
      if (!item.url.startsWith('http')) {
        const url = await this.uploadFile(item.url, 'image/jpeg');
        urls.push(url);
      } else {
        urls.push(item.url);
      }
    }
    return urls;
  },

  // 上传图片列表
  uploadImageList: async function(list) {
    const urls = [];
    for (const url of list) {
      if (!url.startsWith('http')) {
        const uploadedUrl = await this.uploadFile(url, 'image/jpeg');
        urls.push(uploadedUrl);
      } else {
        urls.push(url);
      }
    }
    return urls;
  }
});

## 十、常见问题排查

| 问题 | 可能原因 | 解决方案 |
|------|---------|---------|
| 401 Unauthorized | Token 过期或无效 | 清除本地存储，重新登录 |
| 404 Not Found | 接口路径错误 | 检查 URL 拼接是否正确 |
| CORS 错误 | 后端未配置跨域 | 在微信开发者工具中勾选"不校验合法域名" |
| 图片无法上传 | 上传接口错误 | 检查 `wx.uploadFile` 的 URL |
